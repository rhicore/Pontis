# VFS 增量更新方案

## 核心问题

当前实现是"全量生成"模式，每次重新运行会：
1. 跳过已存在的节点（通过检查特定字段如 `cardinality`）
2. 不会检测源文件变化
3. 不会清理已删除的源文件对应的VFS节点
4. DB表结构变化不会同步

## 解决方案架构

### 1. 增加版本追踪字段

每个 `_meta.yml` 增加 `source_version` 字段：

```yaml
# 文件级节点 (.db/.csv/.json 等)
path: str
modified_at: str              # 源文件mtime
created_at: str
source_version: str           # 新增: 文件hash或版本标识
file_size: int                # 用于快速对比

# 表级节点 (.table)
source_table: str             # 表名
source_version: str           # 新增: schema_hash + row_count + row_hash
row_count: int
column_count: int

# 列级节点 (.col)
source_version: str           # 新增: cardinality + sample_hash
```

### 2. 新增 `sync.py` 模块

职责：检测变化、标记操作、清理孤儿

```python
class ChangeDetector:
    def detect(self, source_path, storage) -> Changeset:
        """返回需要执行的操作列表"""
        
class Changeset:
    create: List[NodeRef]      # 新增节点
    update: List[NodeRef]      # 修改节点  
    delete: List[NodeRef]      # 删除节点（源文件已不存在）
```

### 3. 不同文件类型的检测策略

#### 数据库 (.db)

```python
def get_db_version(db_path):
    """生成数据库版本标识"""
    conn = sqlite3.connect(db_path)
    cursor = conn.cursor()
    
    # 1. 获取所有表的schema
    cursor.execute("SELECT name, sql FROM sqlite_master WHERE type='table'")
    tables = cursor.fetchall()
    schema_str = "|".join(f"{n}:{s}" for n, s in sorted(tables))
    schema_hash = hashlib.md5(schema_str.encode()).hexdigest()[:16]
    
    # 2. 获取总行数
    total_rows = sum(
        cursor.execute(f'SELECT COUNT(*) FROM "{t[0]}"').fetchone()[0]
        for t in tables
    )
    
    conn.close()
    return f"{schema_hash}:{total_rows}"
```

**增量逻辑：**
- schema_hash 变化 → 重建表结构（删除旧.table/.col，创建新的）
- row_count 变化 → 更新统计数据（stats/sample/topk）

#### CSV/TSV

```python
def get_csv_version(csv_path):
    """CSV版本 = 文件hash前16位 + 行数"""
    stat = os.stat(csv_path)
    # 快速方案：mtime + size
    return f"{stat.st_mtime}:{stat.st_size}"
    
    # 精确方案：前N行内容hash
    with open(csv_path, 'rb') as f:
        header = f.read(8192)  # 只读前8KB
        file_hash = hashlib.md5(header).hexdigest()[:16]
    return f"{file_hash}:{stat.st_size}"
```

#### JSON/YAML/XML 等

```python
def get_file_version(file_path):
    """通用文件版本 = mtime:size"""
    stat = os.stat(file_path)
    return f"{int(stat.st_mtime)}:{stat.st_size}"
```

### 4. 更新流程改造

```python
def extract_incremental(target: str, storage: VFSStorage):
    """增量提取流程"""
    
    # Phase 0: 检测变化
    changeset = ChangeDetector().detect(target, storage)
    
    # Phase 1: 清理已删除的节点
    for node in changeset.delete:
        storage.delete_node(node)  # 递归删除节点及其子节点
    
    # Phase 2: 处理新增节点（走原skeleton逻辑）
    for node in changeset.create:
        generate_skeleton_for_file(node.source_path, storage)
    
    # Phase 3: 处理更新节点
    for node in changeset.update:
        if node.type == 'DB':
            update_db_node(node, storage)
        elif node.type in ['CSV', 'TSV']:
            update_csv_node(node, storage)
    
    # Phase 4: 重新生成统计/采样/语义（仅对有变化的节点）
    generate_stats_for_nodes(changeset.create + changeset.update, storage)


def update_db_node(node: NodeRef, storage: VFSStorage):
    """更新数据库节点"""
    old_meta = storage.read_meta(node)
    old_version = old_meta.get('source_version', '')
    
    # 获取新版本
    db_path = storage.resolve_path(old_meta['path'])
    new_version = get_db_version(db_path)
    
    if old_version == new_version:
        return  # 无变化
    
    # 解析旧版本
    old_schema_hash, old_row_count = old_version.split(':')
    new_schema_hash, new_row_count = new_version.split(':')
    
    if old_schema_hash != new_schema_hash:
        # 表结构变化：重建所有子节点
        storage.delete_children(node)  # 删除所有.table/.col/.view
        _expand_database(db_path, node, storage)  # 重新展开
    else:
        # 仅数据变化：更新统计
        for table_node in storage.list_children(node):
            if table_node.name.endswith('.table'):
                update_table_stats(table_node, storage)


def update_csv_node(node: NodeRef, storage: VFSStorage):
    """更新CSV节点"""
    old_meta = storage.read_meta(node)
    csv_path = storage.resolve_path(old_meta['path'])
    
    # 检查列结构是否变化（对比表头）
    old_columns = _get_column_names_from_vfs(node, storage)
    new_columns = _get_column_names_from_csv(csv_path)
    
    if set(old_columns) != set(new_columns):
        # 列变化：重建列节点
        storage.delete_children(node)
        _expand_csv(csv_path, node, storage)
    else:
        # 仅数据变化：标记列需要更新统计
        for col_node in storage.list_children(node):
            col_meta = storage.read_meta(col_node)
            col_meta['needs_update'] = True  # 标记待更新
            storage.write_meta(col_node, col_meta)
```

### 5. 存储层增强

```python
class VFSStorage:
    def delete_node(self, node: NodeRef, recursive: bool = True):
        """删除节点及其所有子节点"""
        if recursive:
            for child in self.list_children(node):
                self.delete_node(child, recursive=True)
        shutil.rmtree(node.full_path)
        
    def delete_children(self, node: NodeRef):
        """删除所有子节点，保留自身"""
        for child in self.list_children(node):
            self.delete_node(child, recursive=True)
```

### 6. 命令行接口

```bash
# 全量生成（现有行为）
python -m extractor ./my_data

# 增量同步（新增）
python -m extractor.sync ./my_data

# 强制重新生成特定节点
python -m extractor.sync ./my_data --force "*.db/account.table"

# 干跑模式（只检测变化，不执行）
python -m extractor.sync ./my_data --dry-run
```

### 7. 实现优先级

1. **P0 - 基础版本检测**
   - 为所有 `_meta.yml` 增加 `source_version` 字段
   - 实现 `ChangeDetector.detect()`

2. **P1 - 清理功能**
   - 实现 `storage.delete_node()`
   - 在 extract 流程最后增加清理孤儿节点步骤

3. **P2 - 增量更新**
   - 实现 `update_db_node()` 处理DB变化
   - 实现 `update_csv_node()` 处理CSV变化

4. **P3 - 优化**
   - DB的 row_hash 检测（只更新真正变化的表）
   - 并行处理多个节点的统计生成

```python
def _cleanup_orphan_nodes(source_path, storage):
    """删除源文件已不存在的VFS节点"""
    for node in storage.find_all_nodes():
        meta = storage.read_meta(node)
        if meta:
            rel_path = meta.get('path')
            if rel_path:
                full_path = os.path.join(source_path, rel_path)
                if not os.path.exists(full_path):
                    storage.delete_node(node)
```
